   127→def head_text(p, n=110):

   129→    t = a.get('contentPlain') or a.get('content') or ''

   133→        try: t = json.dumps(t, ensure_ascii=False)
   134→        except Exception: t = str(type(t))
   135→    t = re.sub(r'<[^>]+>', ' ', t)
   136→    t = re.sub(r'\s+', ' ', t).strip()
   137→    return t[:n]
