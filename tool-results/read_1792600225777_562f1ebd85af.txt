    50→def get(url, name, validate_json=True):

    52→    if os.path.exists(path):
    53→        print(f'SKIP {name}'); return True
    54→    try:
    55→        req = urllib.request.Request(url, headers=UA)
    56→        with urllib.request.urlopen(req, timeout=30) as r:
    57→            body = r.read()
    58→        open(path, 'wb').write(body)
    59→        if validate_json:
    60→            json.loads(body)
    61→        print(f'OK {name} ({len(body)}B)')
    62→        return True
    63→    except Exception as e:
    64→        print(f'FAIL {name}: {e}')
    65→        return False
