    49→def get(url, name, validate_json=True):
    50→    path = os.path.join(OUT, name)
    51→    if os.path.exists(path):
    52→        print(f'SKIP {name}'); return True
    53→    try:
    54→        req = urllib.request.Request(url, headers=UA)
    55→        with urllib.request.urlopen(req, timeout=30) as r:
    56→            body = r.read()
    57→        open(path, 'wb').write(body)
    58→        if validate_json:
    59→            json.loads(body)
    60→        print(f'OK {name} ({len(body)}B)')
    61→        return True
    62→    except Exception as e:
    63→        print(f'FAIL {name}: {e}')
    64→        return False

    97→def head_text(p, n=110):
    98→    a = p.get('attributes', {})
    99→    t = a.get('contentPlain') or a.get('content') or ''
   100→    if not t:
   101→        t = a.get('contentHtml') or ''
   102→    if not isinstance(t, str):
   103→        try: t = json.dumps(t, ensure_ascii=False)
   104→        except Exception: t = str(type(t))
   105→    t = re.sub(r'<[^>]+>', ' ', t)
   106→    t = re.sub(r'\s+', ' ', t).strip()
   107→    return t[:n]
