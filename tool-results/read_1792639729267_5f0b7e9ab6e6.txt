    65→def get(url, name, validate_json=True):
    66→    path = os.path.join(OUT, name)
    67→    if os.path.exists(path):
    68→        print(f'SKIP {name}'); return True
    69→    try:
    70→        req = urllib.request.Request(url, headers=UA)
    71→        with urllib.request.urlopen(req, timeout=30) as r:
    72→            body = r.read()
    73→        open(path, 'wb').write(body)
    74→        if validate_json:
    75→            json.loads(body)
    76→        print(f'OK {name} ({len(body)}B)')
    77→        return True
    78→    except Exception as e:
    79→        print(f'FAIL {name}: {e}')
    80→        return False
