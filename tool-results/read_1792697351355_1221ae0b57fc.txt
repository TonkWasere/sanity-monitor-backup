   108→def get(url, name, validate_json=True):
   109→    path = os.path.join(OUT, name)
   110→    if os.path.exists(path):
   111→        print(f'SKIP {name}'); return True
   112→    try:
   113→        req = urllib.request.Request(url, headers=UA)
   114→        with urllib.request.urlopen(req, timeout=30) as r:
   115→            body = r.read()
   116→        open(path, 'wb').write(body)
   117→        if validate_json:
   118→            json.loads(body)
   119→        print(f'OK {name} ({len(body)}B)')
   120→        return True
   121→    except Exception as e:
   122→        print(f'FAIL {name}: {e}')
   123→        return False
